\newcommand{\featheredge}[1]{%
  \fill[white, path fading=introedgefade, fit fading=true]
    (0,0) circle (#1+0.01);}

% Circle-cropped image (center crop, covers the circle):
% \circleclip{radius-cm}{file}
\newcommand{\circleclip}[2]{%
  \pgfmathsetmacro{\ccd}{2*#1}%
  \begin{scope}
    \clip (0,0) circle (#1);
    \sbox0{\includegraphics{#2}}%
    \ifdim\wd0<\ht0
      \node[inner sep=0pt] at (0,0) {\includegraphics[width=\ccd cm]{#2}};%
    \else
      \node[inner sep=0pt] at (0,0) {\includegraphics[height=\ccd cm]{#2}};%
    \fi
  \end{scope}}

% Portrait/porthole with circle crop + fade: \biopic[radius-cm]{file}
\newcommand{\biopic}[2][2.0]{%
  \begin{tikzpicture}
    \circleclip{#1}{#2}
    \featheredge{#1}
  \end{tikzpicture}}

% Rounded-corner card for rectangular images whose edges carry content
% (screenshots, diagrams, paper titles): \previewcard[width-cm]{file}
\newcommand{\previewcard}[2][7.0]{%
  \sbox0{\includegraphics[width=#1cm]{#2}}%
  \pgfmathsetmacro{\pch}{#1*\ht0/\wd0}%
  \begin{tikzpicture}
    \begin{scope}
      \clip[rounded corners=9pt] (0,0) rectangle (#1,\pch);
      \node[anchor=south west, inner sep=0pt] at (0,0) {\usebox0};
    \end{scope}
  \end{tikzpicture}}
