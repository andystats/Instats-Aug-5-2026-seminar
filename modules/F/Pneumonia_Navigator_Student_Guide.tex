\begin{frame}{Checkpoint 4: target before method}
  \frametag{Target + method}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      {\small\bfseries\color{navorange}CAUSAL TARGET}
      \[
        E[Y^{1}] - E[Y^{0}]
      \]

      {\small\bfseries\color{navorange}STATISTICAL ESTIMAND}
      {\small
      \[
        \begin{aligned}
          \psi &= E_W\{Q(1,W)-Q(0,W)\},\\
          Q(a,W) &= E(Y\mid A=a,W).
        \end{aligned}
      \]}

      \decisionbox{Link}{
        Which Step 4 assumptions identify this target?}
    \end{column}
    \begin{column}{0.44\textwidth}
      {\small\bfseries\color{navorange}PROPOSED ESTIMATOR}

      \vspace{0.1cm}
      {\fontsize{24}{28}\selectfont\bfseries\color{resultgreen}TMLE}

      \begin{itemize}
        \item \(\bar Q(A,W)\): outcome regression
        \item \(g(W)\): treatment mechanism
        \item influence-function interval
      \end{itemize}

      \decisionbox{Boundary}{
        What can TMLE reduce---and what can it not repair?}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Checkpoint 4: plan how the conclusion could fail}
  \frametag{Sensitivity}
  \small
  {\setlength{\tabcolsep}{3pt}%
  \begin{tabular}{@{}>{\bfseries\color{navorange}}L{0.24\linewidth}L{0.70\linewidth}@{}}
    Unmeasured confounding &
      Could residual frailty or care seeking reverse the direction?\\[0.26cm]
    \hline\\[-0.26cm]
    Outcome error &
      Could pneumonia coding change direction or magnitude?\\[0.26cm]
    \hline\\[-0.26cm]
    Selection &
      Could death, disenrollment, or EHR capture be informative?\\[0.26cm]
    \hline\\[-0.26cm]
    Estimation &
      Do learners or propensity truncation change the result?
  \end{tabular}}

  \vspace{0.28cm}
  \decisionbox{Choose one gap}{
    Specify one analysis or source of evidence. A confidence interval does not
    measure these causal gaps.}
\end{frame}

\begin{frame}{Checkpoint 5: write a disciplined conclusion}
  \frametag{After result reveal}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{resultgreen}PROPOSED STATEMENT A}

      \vspace{0.12cm}
      {\large\bfseries\itshape
        ``In this synthetic exercise, adjustment reverses the crude association
        under the stipulated causal model.''}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{resultred}PROPOSED STATEMENT B}

      \vspace{0.12cm}
      {\large\bfseries\itshape\color{resultred}
        ``The vaccine prevents pneumonia in real patients.''}
    \end{column}
  \end{columns}

  \vspace{0.55cm}
  \decisionbox{Accept, correct, or reject}{
    For each claim, cite one assumption or limitation. Rewrite the weaker claim
    so it says no more than the design supports.}
\end{frame}
