\begin{frame}{1. Start with the causal question}

  \begin{center}

      Among adults in a synthetic health-system cohort, what is the effect of
      pneumococcal vaccination at baseline, versus no vaccination, on
      12-month pneumonia hospitalization?}
  \end{center}

  \vspace{0.25cm}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \evidencebox{Observed association}{
        Vaccinated risk: \textcolor{resultred}{\textbf{7.19\%}} \qquad
        Unvaccinated risk: \textbf{6.43\%}\\[3pt]
        Crude risk difference: \textcolor{resultred}{\textbf{\(+0.76\) percentage points}}}
    \end{column}
    \begin{column}{0.44\textwidth}
      \decisionbox{Before interpreting it}{
        Write the population, treatment, comparator, time zero, outcome
        window, and causal contrast you need.}
    \end{column}
  \end{columns}

  \syntheticnote
\end{frame}

\begin{frame}{Checkpoint 4: target before method}
  \frametag{Target + method}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{navorange}CAUSAL TARGET}
      \[
        E[Y^{1}] - E[Y^{0}]
      \]

      {\small\bfseries\color{navorange}STATISTICAL ESTIMAND}
      {\small
      \[
        \begin{aligned}
        E_W\big[&E(Y\mid A=1,W)\\
                &-E(Y\mid A=0,W)\big]
        \end{aligned}
      \]}

      \decisionbox{Link}{
        Which Step 4 assumptions make these quantities coincide?}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{navorange}PROPOSED ESTIMATOR}

      \vspace{0.1cm}
      {\fontsize{30}{34}\selectfont\bfseries\color{resultgreen}TMLE}

      \begin{itemize}
        \item \(\bar Q(A,W)\): outcome regression
        \item \(g(W)\): treatment mechanism
        \item influence-function interval
      \end{itemize}

      \decisionbox{Boundary}{
        Name one problem TMLE can reduce and one problem it cannot repair.}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Checkpoint 4 continued: plan how the conclusion could fail}
  \frametag{Sensitivity}
  \small
  \begin{tabular}{>{\bfseries\color{navorange}}p{0.28\textwidth}p{0.62\textwidth}}
    Unmeasured confounding &
      Could residual frailty or healthcare-seeking change the direction?\\[0.42cm]
    \hline\\[-0.30cm]
    Outcome error &
      Would imperfect pneumonia coding change the direction or magnitude?\\[0.42cm]
    \hline\\[-0.30cm]
    Selection &
      What if death, disenrollment, or EHR capture is informative?\\[0.42cm]
    \hline\\[-0.30cm]
    Estimation &
      Do alternative learners or propensity truncation change the result?
  \end{tabular}

  \vspace{0.3cm}
  \decisionbox{Choose one gap}{
    Specify one sensitivity analysis or one new source of evidence.
    A confidence interval does not measure these causal gaps.}
\end{frame}

\begin{frame}{Checkpoint 5: disciplined conclusion}

        under the stipulated causal model.''}
    \end{column}
    \begin{column}{0.48\textwidth}
      {\small\bfseries\color{resultred}PROPOSED STATEMENT B}

      \vspace{0.12cm}
      {\large\bfseries\itshape\color{resultred}
        ``The vaccine prevents pneumonia in real patients.''}
    \end{column}
  \end{columns}

  \vspace{0.55cm}
  \decisionbox{Accept, correct, or reject}{
    For each statement, cite one assumption or limitation. Rewrite the weaker
    statement so it says no more than the design supports.}
\end{frame}
