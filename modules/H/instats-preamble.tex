        ([xshift=0.22cm]current page.north west);
      \fill[accent]
        ([xshift=0.22cm]current page.south west) rectangle
        ([xshift=0.31cm]current page.north west);

      \node[anchor=north east]
        at ([xshift=-0.65cm,yshift=-0.5cm]current page.north east)
        {\includegraphics[width=2.85cm]{art/InStats_RWL.png}};

      \node[anchor=west,text=slate,text width=12.1cm,align=left,
            font=\bfseries\fontsize{29}{33}\selectfont]
        at ([xshift=1.45cm,yshift=1.48cm]current page.west)
        {\inserttitle};
      \node[anchor=west,text=accent,text width=11.6cm,align=left,
            font=\fontsize{15}{19}\selectfont]

        {\insertsubtitle};

      \draw[black!16,line width=0.5pt]
        ([xshift=1.47cm,yshift=-0.08cm]current page.west) --

      \node[anchor=west,text=slate,

      ([xshift=6.18cm,yshift=-0.42cm]number.east);
