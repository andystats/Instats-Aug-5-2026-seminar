\begin{frame}[t]{One question, one defensible chain}
  \vspace*{0.28cm}

  \begin{enumerate}
    \setlength{\itemsep}{0.26cm}
    \item \textbf{Start with the causal question and target trial.}\\
      Define the population, strategies, outcome, time zero, follow-up, and
      contrast before choosing a model.
    \item \textbf{Make the assumptions visible.}\\
      Connect the causal model and data to identification conditions and the
      adjustment strategy.
    \item \textbf{Match the estimand, estimator, and diagnostics.}\\
      Estimate the quantity you defined; inspect positivity, model behavior,
      uncertainty, and sensitivity.
    \item \textbf{Keep the interpretation inside the evidence.}\\
      State what the estimate supports, what it does not, and the evidence that
      would make you revise the claim.
  \end{enumerate}

  \vspace{0.04cm}
  \begin{center}
    {\scriptsize\color{accent}\bfseries AI can assist with drafting and challenge;
    people remain responsible for the science and the claim.}
  \end{center}
  \note{Recap the four links once. Avoid reopening estimator details. Emphasize
    that the roadmap is a connected decision record, not a checklist completed
    after the analysis.}
\end{frame}
